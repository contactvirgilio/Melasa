cmake_minimum_required(VERSION 3.18)
project(melasa)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(oboe REQUIRED CONFIG)
add_library(melasa SHARED engine.cpp)
target_compile_options(melasa PRIVATE -O2 -ffast-math -Wall)
target_link_libraries(melasa oboe::oboe log)
target_link_options(melasa PRIVATE "-Wl,-z,max-page-size=16384")
